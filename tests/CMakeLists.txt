function(ldds_add_test name)
  add_executable(test_${name} test_${name}.cpp test_main.cpp)
  target_link_libraries(test_${name} PRIVATE ldds::ldds ldds_warnings)
  add_test(NAME ${name} COMMAND test_${name})
  set_tests_properties(${name} PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 60)
endfunction()

ldds_add_test(serialization)
ldds_add_test(wire)
ldds_add_test(discovery)
ldds_add_test(pubsub)
ldds_add_test(reliable)
ldds_add_test(deadline)
ldds_add_test(capture)
ldds_add_test(udp)
